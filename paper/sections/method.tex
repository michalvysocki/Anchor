\paragraph{A family of multi-step targets.}
For a horizon $k = p + q$, the target with bootstrap point $p$ is
\begin{equation}
  \Tp = g^{q}\big(\varphi(y_{\le t+p})\big), \qquad p = 0, \dots, k.
\end{equation}
\todo{$p = k$: Monte Carlo (encoded true future, as in TD-MPC and SPR); $0 < p < k$: bootstrapped (TD networks); a learned head as the amortized bootstrapped target.}

\paragraph{Path disagreement.}
\begin{equation}
  \gap_{p,q}(\z{t}) = \E\big[\E[y_{t+k} \mid \z{t+p}] \mid \z{t}\big] - \E[y_{t+k} \mid \z{t}].
\end{equation}
\todo{Zero for a Markov latent; observable without the state; how it is estimated (open: separating it from target variance).}

\paragraph{The Anchor loss.}
\begin{equation}
  \Lanchor = \Lone + \lambda \sum_{p \ge 2} w_p \, \big\| g^{k}(\z{t}) - \sg[\Tp] \big\|^2 .
\end{equation}
A fixed $\lambda$ suffices: the gradient of the Anchor term vanishes when the latent is Markov (Proposition~\ref{prop:t1}), and bootstrap points $p = 1$ carry no signal at the one-step optimum (Remark~\ref{rem:p1}). \todo{Weights $w_p$ from T3; grounding through the one-step loss and real-observation targets; stop-gradient; adaptive $\lambda$ only if experiments show harm where Proposition~\ref{prop:t1} does not apply.}
